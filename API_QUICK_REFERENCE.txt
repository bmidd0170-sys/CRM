// ============================================
// VALIDATION RULES SUMMARY
// ============================================

ADMIN:
  - name: required, max 100 chars
  - email: required, valid email, unique
  - role: required, enum (Super Admin, Admin, Manager, Viewer)
  - restrictions: optional, array
  - online: optional, boolean
  - changes: optional, array

DONOR:
  - name: required, max 100 chars
  - email: required, valid email, unique
  - total: optional, non-negative number
  - lastDonation: optional, ISO 8601 datetime
  - status: required, enum (Active, Inactive, Pending)
  - tags: optional, array

CAMPAIGN:
  - name: required, max 200 chars
  - goal: required, positive number
  - raised: optional, non-negative number
  - startDate: required, ISO 8601 datetime
  - endDate: required, ISO 8601 datetime (must be after startDate)
  - description: required

EVENT:
  - name: required, max 200 chars
  - date: required, ISO 8601 datetime
  - description: required
  - image: optional, valid URL
  - campaignId: optional, valid campaign ID

DONATION:
  - amount: required, positive number
  - date: required, ISO 8601 datetime
  - donorId: required, valid donor ID
  - campaignId: optional, valid campaign ID

NOTIFICATION:
  - type: required, enum (Info, Warning, Error, Success)
  - message: required, max 500 chars
  - date: required, ISO 8601 datetime
  - read: optional, boolean

LOGIN:
  - email: required, valid email
  - password: required, min 6 chars


// ============================================
// ERROR CODES
// ============================================

200 OK - Success
201 Created - Resource created successfully
400 Bad Request - Validation failed or missing required fields
404 Not Found - Resource not found
409 Conflict - Unique constraint violated (e.g., email exists)
500 Internal Server Error - Server error


// ============================================
// JAVASCRIPT FETCH EXAMPLES
// ============================================

// GET request
const data = await fetch('/api/donors').then(r => r.json());

// POST request with validation
const response = await fetch('/api/donors', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    name: 'John Doe',
    email: 'john@example.com',
    status: 'Active'
  })
});

if (response.ok) {
  const donor = await response.json();
  console.log('Created:', donor);
} else if (response.status === 400) {
  const error = await response.json();
  console.error('Validation error:', error.details);
} else if (response.status === 409) {
  console.error('Email already exists');
}

// PUT request
const response = await fetch('/api/donors', {
  method: 'PUT',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    id: 1,
    status: 'Inactive'
  })
});

// DELETE request
const response = await fetch('/api/donors?id=1', {
  method: 'DELETE'
});
